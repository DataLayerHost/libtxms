cmake_minimum_required(VERSION 3.16)
project(txms VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
set(CMAKE_POSITION_INDEPENDENT_CODE ON CACHE BOOL "Build position independent code")
option(TXMS_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(TXMS_FUZZ "Build libFuzzer target (Clang)" OFF)
add_library(txms src/txms.c src/txms_hex.c src/txms_utf16.c src/txms_encode.c src/txms_decode.c)
target_include_directories(txms PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(txms PUBLIC c_std_11)
if(MSVC)
	target_compile_options(txms PRIVATE /W4 /WX /utf-8)
	set_target_properties(txms PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
	target_compile_options(txms PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set_target_properties(txms PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(TXMS_SANITIZE)
	target_compile_options(txms PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
	target_link_options(txms PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
	add_executable(txms_test tests/test.c)
	target_link_libraries(txms_test PRIVATE txms)
	if(MSVC)
		target_compile_options(txms_test PRIVATE /W4 /WX /utf-8 /UNDEBUG)
	else()
		target_compile_options(txms_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
	endif()
	add_test(NAME txms_unit COMMAND txms_test)
	add_executable(txms_oracle tests/oracle.c)
	target_link_libraries(txms_oracle PRIVATE txms)
endif()
if(TXMS_FUZZ)
	target_compile_options(txms PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
	add_executable(txms_fuzz tests/fuzz.c)
	target_link_libraries(txms_fuzz PRIVATE txms)
	target_compile_options(txms_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
	target_link_options(txms_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS txms EXPORT txmsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/txms.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT txmsTargets NAMESPACE txms:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/txms)
file(RELATIVE_PATH TXMS_PC_PREFIX_RELATIVE "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
configure_file(cmake/txms.pc.in txms.pc @ONLY)
configure_package_config_file(cmake/txmsConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/txmsConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/txms)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/txmsConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/txmsConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/txmsConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/txms)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/txms.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
